cmake_minimum_required(VERSION 3.20)
project(native_ui_controls LANGUAGES C CXX)
find_package(native_ui CONFIG REQUIRED)
set(UI_BACKEND sdl3 CACHE STRING "sdl3, opengl or raylib")
set(UI_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/${UI_BACKEND}/c/showcase.c" CACHE FILEPATH "Application or test harness")
set(UI_OUTPUT "" CACHE FILEPATH "Optional executable destination")
add_executable(showcase "${UI_SOURCE}")
if(UI_CPP)
    set_source_files_properties("${UI_SOURCE}" PROPERTIES LANGUAGE CXX)
endif()
target_compile_features(showcase PRIVATE c_std_11 cxx_std_17)
target_link_libraries(showcase PRIVATE native_ui::native_ui)
if(UI_BACKEND STREQUAL "raylib")
    find_package(raylib CONFIG REQUIRED)
    target_link_libraries(showcase PRIVATE raylib)
    if(APPLE)
        target_link_libraries(showcase PRIVATE "-framework IOKit" "-framework Cocoa" "-framework OpenGL")
    elseif(UNIX)
        target_link_libraries(showcase PRIVATE m pthread dl)
    endif()
else()
    find_package(SDL3 CONFIG REQUIRED)
    target_link_libraries(showcase PRIVATE SDL3::SDL3)
    if(UI_BACKEND STREQUAL "opengl")
        find_package(OpenGL REQUIRED)
        target_link_libraries(showcase PRIVATE OpenGL::GL)
        target_compile_definitions(showcase PRIVATE GL_SILENCE_DEPRECATION)
    elseif(NOT UI_BACKEND STREQUAL "sdl3")
        message(FATAL_ERROR "Unknown UI_BACKEND: ${UI_BACKEND}")
    endif()
endif()
if(MSVC)
    target_compile_options(showcase PRIVATE /W3 /WX /utf-8)
    target_compile_definitions(showcase PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(showcase PRIVATE -Wall -Wextra -Werror)
endif()
if(UI_OUTPUT)
    get_filename_component(_dir "${UI_OUTPUT}" DIRECTORY)
    get_filename_component(_name "${UI_OUTPUT}" NAME_WE)
    set_target_properties(showcase PROPERTIES OUTPUT_NAME "${_name}" RUNTIME_OUTPUT_DIRECTORY "${_dir}")
    foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(showcase PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${_dir}")
    endforeach()
endif()
if(APPLE)
    set_target_properties(showcase PROPERTIES BUILD_RPATH "@loader_path")
elseif(UNIX)
    set_target_properties(showcase PROPERTIES BUILD_RPATH "$ORIGIN")
endif()

# Windows does not search link-time library directories for DLLs at runtime.
if(WIN32)
    foreach(runtime native_ui::native_ui SDL3::SDL3 raylib)
        if(TARGET ${runtime})
            get_target_property(kind ${runtime} TYPE)
            if(kind STREQUAL "SHARED_LIBRARY")
                add_custom_command(TARGET showcase POST_BUILD
                    COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "$<TARGET_FILE:${runtime}>" "$<TARGET_FILE_DIR:showcase>" VERBATIM)
            endif()
        endif()
    endforeach()
endif()
